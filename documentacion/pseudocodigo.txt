PROCESO PRINCIPAL
  INTENTAR obtener bloqueo de sesión
  registros <- CARGAR(ruta)
  SI hay error ENTONCES informar, liberar y finalizar
  MOSTRAR_INTERFAZ(registros)
  Esperar eventos de Tkinter hasta cerrar la ventana
  Al cerrar: liberar bloqueo y finalizar
FINPROCESO

PROCEDIMIENTO MOSTRAR_INTERFAZ(registros)
  Mostrar formulario, búsqueda, lista y estado
  Asociar botones a REGISTRAR, BUSCAR, ORDENAR y CAMBIAR_ESTADO
  Mostrar ORDENAR(registros, "codigo")
FINPROCEDIMIENTO

FUNCIÓN VALIDAR(tipo, fecha, asunto, descripción, referencia)
  Verificar que todos los campos sean cadenas y quitar espacios externos
  Verificar tipo permitido y fecha real con formato AAAA-MM-DD
  Verificar 3 <= longitud(asunto) <= 100
  Verificar 10 <= longitud(descripción) <= 1000
  Verificar longitud(referencia) <= 60
  Si hay error: lanzar error con explicación
  Retornar campos normalizados
FINFUNCIÓN

PROCEDIMIENTO REGISTRAR
  Leer formulario
  Si no se confirma que son datos ficticios: avisar y retornar
  INTENTAR
    datos <- VALIDAR(campos)
    mayor <- 0
    Para cada registro: mayor <- máximo(mayor, número de su código)
    código <- "IS-" + (mayor + 1 con mínimo seis dígitos)
    nuevo <- datos con código, estado Pendiente y ficticio Verdadero
    GUARDAR(ruta, registros + [nuevo])
    Añadir nuevo a registros
    Limpiar formulario y mostrar listado ordenado
    Mostrar código y confirmación
  SI hay error: mostrarlo y conservar la lista anterior
FINPROCEDIMIENTO

FUNCIÓN BUSCAR(registros, código)
  código <- mayúsculas y sin espacios externos
  Para cada registro en registros:
    Si registro.código = código: retornar registro
  Retornar Nulo
FINFUNCIÓN

FUNCIÓN ORDENAR(registros, criterio)
  Verificar criterio: codigo o fecha
  resultado <- copia de registros
  clave <- número del código, o pareja (fecha, número del código)
  Para i desde 1 hasta longitud(resultado)-1:
    actual <- resultado[i]; j <- i-1
    Mientras j >= 0 Y clave(resultado[j]) > clave(actual):
      resultado[j+1] <- resultado[j]; j <- j-1
    resultado[j+1] <- actual
  Retornar resultado
FINFUNCIÓN

PROCEDIMIENTO GUARDAR(ruta, lista)
  Validar estructura, campos, códigos únicos y marca ficticia
  Escribir lista como JSON UTF-8 en un archivo temporal
  Vaciar búfer y sincronizar escritura
  Reemplazar archivo destino con el temporal
  Si falla: propagar error; conservar destino anterior
  Eliminar temporal restante
FINPROCEDIMIENTO

FUNCIÓN CARGAR(ruta)
  Si no existe el archivo: retornar lista vacía
  Leer JSON y validar estructura, campos, códigos y marca ficticia
  Retornar lista; si hay error, propagarlo sin sobrescribir
FINFUNCIÓN

PROCEDIMIENTO CAMBIAR_ESTADO(código, estado)
  Exigir selección de fila, código existente y estado permitido
  copia <- copiar cada diccionario de registros
  Actualizar estado en copia
  GUARDAR(ruta, copia)
  Reemplazar registros en memoria por copia y mostrar listado
  Si falla: informar y conservar datos anteriores
FINPROCEDIMIENTO
